# Phase 1 Day 2+ deep-learning extras.
#
# Install AFTER the base requirements.txt:
#     pip install -r requirements-deep.txt
#
# DO NOT use -U here. -U force-upgrades EVERY package listed, including
# torch. Colab pre-installs a torch + torchvision pair compiled against
# the same CUDA version (e.g. torch 2.10+cu128 + torchvision 0.25+cu128).
# -U bumps torch to the latest (2.11+cu130) while leaving torchvision at
# cu128, and torchvision's _check_cuda_version() then refuses to load,
# which cascades into peft / transformers import failures (torchvision is
# pulled in transitively via transformers.image_utils).
#
# Without -U, pip leaves already-satisfied packages alone and only upgrades
# the one constraint-violator (torchao 0.10 -> 0.16+). torch + torchvision
# stay paired. peft loads cleanly. This is what we want.
#
# DeepChem is NOT in this file because (a) it has no Python 3.12 wheel on
# PyPI which causes pip to abort the entire resolution before any package
# is installed, and (b) Phase 1 Day 2 doesn't actually use it (Tox21 aux
# head is off). Phase 2 will move Tox21 loading to a wheel-friendly path
# (direct CSV download from MoleculeNet) so we don't need deepchem at all.

torch>=2.1
transformers>=4.40
peft>=0.10
accelerate>=0.27

# torchao>=0.16 is required by recent peft (>=0.18) LoRA dispatcher.
# is_torchao_available() *raises* (rather than returning False) when torchao
# is older than 0.16, killing get_peft_model() before any LoRA adapter is
# created. Pinning here forces the upgrade past Colab's pre-installed 0.10.0.
torchao>=0.16
